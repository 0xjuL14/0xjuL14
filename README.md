    const char *name;
    const char *role;
    const char *languages[3];
} DevInProgress;

void introduce(DevInProgress me);

int main() {
    DevInProgress me = {
        .name = "Julia",
        .role = "ADS, com foco em Cybersecurity",
        .languages = {"pt_BR", "en_US", NULL}
    };

    introduce(me);

    return 0;
}

void introduce(DevInProgress me) {
    printf("Oie! Eu sou a %s!\n", me.name);
    printf("Área de estudo: %s\n", me.role);

    printf("Idiomas: ");
    for (int i = 0; me.languages[i] != NULL; i++) {
        printf("%s ", me.languages[i]);
    }

    printf("\n Obrigada por visitar meu perfi! <3 \n");
}
